#include <iostream>
#include <ctime>
#include <cstdlib>
#include <iomanip>
#include <chrono>

using namespace std;

// Variant 2
const int N1 = 800;
const int N2 = 3000;
const int N3 = 7000;

const int ITERATIONS = 100;

// ============================================================
// SIMPLE INSERTION SORT
// ============================================================
void insertionSort(int arr[], int n) {

for (int i = 1; i < n; i++) {

int key = arr[i];
int j = i - 1;

while (j >= 0 && arr[j] > key) {

arr[j + 1] = arr[j];

j--;
}

arr[j + 1] = key;
}
}

// ============================================================
// GENERATE RANDOM ARRAY
// ============================================================
void generateArray(int arr[], int n) {

for (int i = 0; i < n; i++) {
arr[i] = rand() % 10000;
}
}

// ============================================================
// COPY ARRAY
// ============================================================
void copyArray(int source[], int dest[], int n) {

for (int i = 0; i < n; i++) {
dest[i] = source[i];
}
}

// ============================================================
// CHECK IF ARRAY IS SORTED
// ============================================================
bool isSorted(int arr[], int n) {

for (int i = 0; i < n - 1; i++) {

if (arr[i] > arr[i + 1]) {
return false;
}
}

return true;
}

// ============================================================
// MEASURE SORTING TIME
// ============================================================
double measureTime(int original[], int n) {

int* temp = new int[n];

double totalTime = 0.0;

for (int i = 0; i < ITERATIONS; i++) {

// Copy original array
copyArray(original, temp, n);

// Start timer
auto start = chrono::high_resolution_clock::now();

// Sort
insertionSort(temp, n);

// Stop timer
auto end = chrono::high_resolution_clock::now();

chrono::duration<double> duration = end - start;

totalTime += duration.count();

// Verify sorting
if (!isSorted(temp, n)) {

cout << "ERROR: Array was not sorted!" << endl;

delete[] temp;

return -1;
}
}

delete[] temp;

return totalTime / ITERATIONS;
}

// ============================================================
// MAIN
// ============================================================
int main() {

srand(static_cast<unsigned int>(time(nullptr)));

int sizes[] = {N1, N2, N3};

cout << "" << endl;

cout << " VARIANT 2" << endl;
cout << "" << endl;

cout << "Array sizes: 800, 3000, 7000" << endl;
cout << "Iterations: " << ITERATIONS << endl;

cout << endl;

cout << setw(20) << "Method"

<< setw(15) << "n = 3000"
<< setw(15) << "n = 7000"
<< endl;

cout << "--------------------------------------------------" << endl;

cout << setw(20) << "Insertion Sort";

for (int i = 0; i < 3; i++) {

int n = sizes[i];

int* original = new int[n];

generateArray(original, n);

double averageTime = measureTime(original, n);

cout << setw(15)
<< fixed
<< setprecision(6)
<< averageTime;

delete[] original;
}

cout << endl;

cout << "==================================================" << endl;

cout << "Complexity: O(n^2)" << endl;

cout << "==================================================" << endl;

return 0;
}
