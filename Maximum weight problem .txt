int n = cycle.size(); for (int i = 0; i < n - 1; i++) sum += graph[cycle[i]][cycle[i + 1]]; sum += graph[cycle[n - 1]][cycle[0]]; return sum; } // ------------------------------------------------------------ // HEURISTIC ALGORITHM // Greedy maximum-edge algorithm // ------------------------------------------------------------ vector greedyMaximumCycle(const vector>& graph) { int n = graph.size(); vector bestCycle; long long bestLength = -1; // Try every vertex as a starting vertex. // This improves the greedy solution while keeping it heuristic. for (int start = 0; start < n; start++) { vector used(n, false); vector cycle; int current = start; cycle.push_back(current); used[current] = true; // At every step choose the unvisited vertex // connected by the largest edge. for (int step = 1; step < n; step++) { int next = -1; int maximumWeight = -1; for (int v = 0; v < n; v++) { if (!used[v] && graph[current][v] > maximumWeight) { maximumWeight = graph[current][v]; next = v; } } cycle.push_back(next); used[next] = true; current = next; } long long length = cycleLength(graph, cycle); if (length > bestLength) { bestLength = length; bestCycle = cycle; } } return bestCycle; } // ------------------------------------------------------------ // BRUTE FORCE ALGORITHM // Checks all possible Hamiltonian cycles. // ------------------------------------------------------------ void bruteForceRecursive(const vector>& graph, vector& cycle, vector& used, long long& bestLength, vector& bestCycle) { int n = graph.size(); if ((int)cycle.size() == n) { long long length = cycleLength(graph, cycle); if (length > bestLength) { bestLength = length; bestCycle = cycle; } return; } int current = cycle.back(); for (int v = 1; v < n; v++) { if (!used[v]) { used[v] = true; cycle.push_back(v); bruteForceRecursive(graph, cycle, used, bestLength, bestCycle); cycle.pop_back(); used[v] = false; } } } vector bruteForceMaximumCycle(const vector>& graph) { int n = graph.size();
