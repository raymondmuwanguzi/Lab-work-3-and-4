const int N1 = 800;
const int N2 = 3000;

const int ITERATIONS = 100;

// ============================================================
// SHELL SORT
// ============================================================
void shellSort(int arr[], int n) {
// Start with a large gap, then reduce the gap
for (int gap = n / 2; gap > 0; gap /= 2) {

// Gapped insertion sort
for (int i = gap; i < n; i++) {

int temp = arr[i];
int j;

for (j = i;
j >= gap && arr[j - gap] > temp;
j -= gap) {

arr[j] = arr[j - gap];
}

arr[j] = temp;
}
}
}

// ============================================================
// SIMPLE INSERTION SORT
// ============================================================
void insertionSort(int arr[], int n) {

for (int i = 1; i < n; i++) {

int key = arr[i];
int j = i - 1;

while (j >= 0 && arr[j] > key) {

arr[j + 1] = arr[j];
j--;
}

arr[j + 1] = key;
}
}

// ============================================================
// GENERATE RANDOM ARRAY
// ============================================================
void generateArray(int arr[], int n) {

for (int i = 0; i < n; i++) {
arr[i] = rand() % 10000;
}
}

// ============================================================
// COPY ARRAY
// ============================================================
void copyArray(int source[], int dest[], int n) {

for (int i = 0; i < n; i++) {
dest[i] = source[i];
}
}

// ============================================================
// CHECK WHETHER ARRAY IS SORTED
// ============================================================
bool isSorted(int arr[], int n) {

for (int i = 0; i < n - 1; i++) {

if (arr[i] > arr[i + 1]) {
return false;
}
}

return true;
}

// ============================================================
// MEASURE SORTING TIME
// ============================================================
double measureTime(
void (*sortFunc)(int[], int),
int original[],
int n,
int iterations
) {

int* temp = new int[n];

double totalTime = 0.0;

for (int iter = 0; iter < iterations; iter++) {

// Copy the original unsorted array
copyArray(original, temp, n);

// Start timer
auto start = chrono::high_resolution_clock::now();

// Sort
sortFunc(temp, n);

// Stop timer
auto end = chrono::high_resolution_clock::now();

chrono::duration<double> duration = end - start;

totalTime += duration.count();

// Verify sorting
if (!isSorted(temp, n)) {

cout << "ERROR: Array was not sorted correctly!"
<< endl;

delete[] temp;

return -1.0;
}
}

delete[] temp;

// Return average time
return totalTime / iterations;
}

// ============================================================
// TEST ONE SORTING METHOD
// ============================================================
void testMethod(
const string& methodName,
void (*sortFunc)(int[], int),
int sizes[],
int numberOfSizes
) {

cout << setw(20) << left << methodName;

for (int i = 0; i < numberOfSizes; i++) {

int n = sizes[i];

// Create original random array
int* original = new int[n];

generateArray(original, n);

// Measure average sorting time
double averageTime =
measureTime(
sortFunc,
original,
n,
ITERATIONS
);

cout << setw(18)
<< fixed
<< setprecision(6)
<< averageTime;

delete[] original;
}

cout << endl;
}
