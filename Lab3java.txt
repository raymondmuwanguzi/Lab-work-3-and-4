package lab3.model;

/** Base class for every human in the model: name, age and gender. */
public abstract class Person {

    private final String name;
    private final int age;
    private final Gender gender;

    protected Person(String name, int age, Gender gender) {
        if (name == null || name.trim().isEmpty()) {
            throw new IllegalArgumentException("Name must not be empty");
        }
        if (age <= 0) {
            throw new IllegalArgumentException("Age must be positive: " + age);
        }
        if (gender == null) {
            throw new IllegalArgumentException("Gender must not be null");
        }
        this.name = name;
        this.age = age;
        this.gender = gender;
    }

    public String getName() {
        return name;
    }

    public int getAge() {
        return age;
    }

    public Gender getGender() {
        return gender;
    }

    /** Common part of the description used by subclasses: name, age, gender. */
    @Override
    public String toString() {
        return name + ", age " + age + ", " + gender;
    }
}

package lab3.model;

/** A single immutable mark given by a teacher in a given week of the semester. */
public class Grade {

    public static final int MIN_VALUE = 2;
    public static final int MAX_VALUE = 5;

    private final int value;
    private final int week;
    private final Subject subject;
    private final Teacher teacher;

    public Grade(int value, int week, Subject subject, Teacher teacher) {
        if (value < MIN_VALUE || value > MAX_VALUE) {
            throw new IllegalArgumentException(
                    "Grade must be in range " + MIN_VALUE + ".." + MAX_VALUE + ": " + value);
        }
        if (week < 1) {
            throw new IllegalArgumentException("Week number must be >= 1: " + week);
        }
        if (subject == null || teacher == null) {
            throw new IllegalArgumentException("Subject and teacher must not be null");
        }
        this.value = value;
        this.week = week;
        this.subject = subject;
        this.teacher = teacher;
    }

    public int getValue() {
        return value;
    }

    public int getWeek() {
        return week;
    }

    public Subject getSubject() {
        return subject;
    }

    public Teacher getTeacher() {
        return teacher;
    }

    @Override
    public String toString() {
        return "week " + week + ": " + subject + " = " + value;
    }
}

package lab3.model;

/**
 * Status ("mood") of a parent and the size of the bonus (in thousands) paid to the student.
 * <ul>
 * <li>average below 4.0 (task: 3 - 3.9)  -> GLOOMY, no bonus;</li>
 * <li>average 4.0 - 4.5                  -> SATISFIED, 5 thousand;</li>
 * <li>average 4.6 - 5.0 (task: "5.6-5")  -> JOYFUL, 10 thousand.</li>
 * </ul>
 * Averages between 4.5 and 4.6 (not covered by the task) are treated as SATISFIED.
 */
public enum ParentMood {
    NEUTRAL("neutral", 0),     // initial state, before the session results are known
    GLOOMY("gloomy", 0),
    SATISFIED("satisfied", 5),
    JOYFUL("joyful", 10);

    public static final double SATISFIED_FROM = 4.0;
    public static final double JOYFUL_FROM = 4.6;

    private final String title;
    private final int bonus;

    ParentMood(String title, int bonus) {
        this.title = title;
        this.bonus = bonus;
    }

    /** Bonus in thousands. */
    public int getBonus() {
        return bonus;
    }

    /** Determines the mood of a parent by the average grade of the student. */
    public static ParentMood forAverage(double average) {
        if (average >= JOYFUL_FROM) {
            return JOYFUL;
        }
        if (average >= SATISFIED_FROM) {
            return SATISFIED;
        }
        return GLOOMY;
    }

    @Override
    public String toString() {
        return title;
    }
}


6. src/lab3/model/Student.java

package lab3.model;

import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Locale;
import java.util.Map;

/** A student who receives grades in several subjects during the semester. */
public class Student extends Person {

    /** A teacher may give at most this many grades to one student in his subject. */
    public static final int MAX_GRADES_PER_SUBJECT = 10;

    private final Map<Subject, List<Grade>> grades = new LinkedHashMap<Subject, List<Grade>>();
    private Parent parent;
    private int bonus; // in thousands

    public Student(String name, int age, Gender gender) {
        super(name, age, gender);
    }

    public Parent getParent() {
        return parent;
    }

    /** Package-private: the link is maintained by {@link Parent#addChild(Student)}. */
    void setParent(Parent parent) {
        this.parent = parent;
    }

    /** Number of grades received in the subject. */
    public int getGradeCount(Subject subject) {
        List<Grade> list = grades.get(subject);
        return list == null ? 0 : list.size();
    }

    /** True if the student already has a grade in this subject for the given week. */
    public boolean hasGradeInWeek(Subject subject, int week) {
        List<Grade> list = grades.get(subject);
        if (list == null) {
            return false;
        }
        for (Grade g : list) {
            if (g.getWeek() == week) {
                return true;
            }
        }
        return false;
    }

    /**
     * Adds a grade. Enforces the rules of the task: no more than 10 grades per subject
     * and no more than one grade per week per subject.
     */
    public void addGrade(Grade grade) {
        if (grade == null) {
            throw new IllegalArgumentException("Grade must not be null");
        }
        Subject subject = grade.getSubject();
        if (getGradeCount(subject) >= MAX_GRADES_PER_SUBJECT) {
            throw new IllegalStateException(getName() + " already has " + MAX_GRADES_PER_SUBJECT
                    + " grades in " + subject);
        }
        if (hasGradeInWeek(subject, grade.getWeek())) {
            throw new IllegalStateException(getName() + " already has a grade in " + subject
                    + " for week " + grade.getWeek());
        }
        List<Grade> list = grades.get(subject);
        if (list == null) {
            list = new ArrayList<Grade>();
            grades.put(subject, list);
        }
        list.add(grade);
    }

    /** Read-only view of the grades in a subject. */
    public List<Grade> getGrades(Subject subject) {
        List<Grade> list = grades.get(subject);
        if (list == null) {
            return Collections.emptyList();
        }
        return Collections.unmodifiableList(list);
    }

    /** Average grade in one subject (0 if there are no grades). */
    public double getAverageGrade(Subject subject) {
        return average(grades.get(subject));
    }

    /** Average grade for the whole session over all subjects (0 if there are no grades). */
    public double getAverageGrade() {
        List<Grade> all = new ArrayList<Grade>();
        for (List<Grade> list : grades.values()) {
            all.addAll(list);
        }
        return average(all);
    }

    private static double average(List<Grade> list) {
        if (list == null || list.isEmpty()) {
            return 0.0;
        }
        int sum = 0;
        for (Grade g : list) {
            sum += g.getValue();
        }
        return (double) sum / list.size();
    }

    /** Bonus received from the parent, in thousands. */
    public int getBonus() {
        return bonus;
    }

    public void receiveBonus(int amount) {
        if (amount < 0) {
            throw new IllegalArgumentException("Bonus must not be negative: " + amount);
        }
        bonus += amount;
    }

    @Override
    public String toString() {
        return String.format(Locale.US,
                "Student: %s, average grade for the session: %.2f, bonus: %d thousand",
                super.toString(), getAverageGrade(), bonus);
    }
}

package lab3.model;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

/** A parent of one or several students. Pays bonuses depending on the students' results. */
public class Parent extends Person {

    private final List<Student> children = new ArrayList<Student>();
    private ParentMood mood = ParentMood.NEUTRAL;
    private int totalBonusPaid; // in thousands
    private boolean reviewed;

    public Parent(String name, int age, Gender gender) {
        super(name, age, gender);
    }

    public void addChild(Student student) {
        if (student == null) {
            throw new IllegalArgumentException("Student must not be null");
        }
        if (student.getParent() != null) {
            throw new IllegalStateException(student.getName() + " already has a parent");
        }
        children.add(student);
        student.setParent(this);
    }

    public List<Student> getChildren() {
        return Collections.unmodifiableList(children);
    }

    public ParentMood getMood() {
        return mood;
    }

    public int getTotalBonusPaid() {
        return totalBonusPaid;
    }

    /**
     * Reviews the session results: every child gets the bonus corresponding to his own
     * average grade; the parent's status is determined by the mean of the children's averages.
     * Bonuses are paid only once.
     */
    public void reviewResults() {
        if (reviewed || children.isEmpty()) {
            return;
        }
        double sumOfAverages = 0.0;
        for (Student child : children) {
            double average = child.getAverageGrade();
            int bonus = ParentMood.forAverage(average).getBonus();
            child.receiveBonus(bonus);
            totalBonusPaid += bonus;
            sumOfAverages += average;
        }
        mood = ParentMood.forAverage(sumOfAverages / children.size());
        reviewed = true;
    }

    @Override
    public String toString() {
        StringBuilder names = new StringBuilder();
        for (Student child : children) {
            if (names.length() > 0) {
                names.append(", ");
            }
            names.append(child.getName());
        }
        return "Parent: " + super.toString() + ", status: " + mood + ", student(s): "
                + (names.length() == 0 ? "-" : names) + ", bonuses paid: " + totalBonusPaid
                + " thousand";
    }
}

package lab3.model;

/** A teacher who gives grades to students in his own subject. */
public class Teacher extends Person {

    private final Subject subject;

    public Teacher(String name, int age, Gender gender, Subject subject) {
        super(name, age, gender);
        if (subject == null) {
            throw new IllegalArgumentException("Subject must not be null");
        }
        this.subject = subject;
    }

    public Subject getSubject() {
        return subject;
    }

    /** Can the teacher give a grade to the student in the given week (10 max, 1 per week)? */
    public boolean canGrade(Student student, int week) {
        return student.getGradeCount(subject) < Student.MAX_GRADES_PER_SUBJECT
                && !student.hasGradeInWeek(subject, week);
    }

    /**
     * Gives a grade to the student for the teacher's subject.
     *
     * @throws IllegalStateException if the limits of the task are violated
     */
    public Grade grade(Student student, int week, int value) {
        if (student == null) {
            throw new IllegalArgumentException("Student must not be null");
        }
        Grade grade = new Grade(value, week, subject, this);
        student.addGrade(grade);
        return grade;
    }

    @Override
    public String toString() {
        return "Teacher: " + super.toString() + ", subject: " + subject;
    }
}

package lab3.model;

import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Set;

/** The whole model of one semester: a group of students, their parents and teachers. */
public class SemesterModel {

    private final int weeks;
    private final List<Student> students = new ArrayList<Student>();
    private final List<Parent> parents = new ArrayList<Parent>();
    private final List<Teacher> teachers = new ArrayList<Teacher>();

    public SemesterModel(int weeks) {
        if (weeks < Student.MAX_GRADES_PER_SUBJECT) {
            throw new IllegalArgumentException(
                    "A semester must have at least " + Student.MAX_GRADES_PER_SUBJECT + " weeks");
        }
        this.weeks = weeks;
    }

    public int getWeeks() {
        return weeks;
    }

    public void addStudent(Student student) {
        students.add(student);
    }

    public void addParent(Parent parent) {
        parents.add(parent);
    }

    public void addTeacher(Teacher teacher) {
        teachers.add(teacher);
    }

    public List<Student> getStudents() {
        return Collections.unmodifiableList(students);
    }

    public List<Parent> getParents() {
        return Collections.unmodifiableList(parents);
    }

    public List<Teacher> getTeachers() {
        return Collections.unmodifiableList(teachers);
    }

    /** Subjects taught in the semester, in the order of the teachers. */
    public List<Subject> getSubjects() {
        Set<Subject> subjects = new LinkedHashSet<Subject>();
        for (Teacher t : teachers) {
            subjects.add(t.getSubject());
        }
        return new ArrayList<Subject>(subjects);
    }

    /** End of the semester: every parent reviews the results and pays the bonuses. */
    public void settleBonuses() {
        for (Parent p : parents) {
            p.reviewResults();
        }
    }
}

package lab3.controller;

import java.io.BufferedWriter;
import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStreamWriter;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.util.ArrayList;
import java.util.List;
import java.util.logging.FileHandler;
import java.util.logging.Level;
import java.util.logging.LogManager;
import java.util.logging.Logger;
import java.util.logging.SimpleFormatter;

/**
 * Common class for error handling, messages and logging (Appendix 2 of the lab).
 * <ul>
 * <li>keeps the list of all registered errors, so the number of errors is always known;</li>
 * <li>writes the errors to the error log (java.util.logging, configured in logging.properties);</li>
 * <li>writes the operation logs of the experiments (one text file per experiment).</li>
 * </ul>
 */
public class ErrMsgLog {

    /** Folder for all log files. It must be the same as in logging.properties. */
    public static final String LOG_DIR = "logs";
    private static final String CONFIG_RESOURCE = "logging.properties";

    private final List<Exception> errList = new ArrayList<Exception>();
    private final Logger log;
    private FileHandler fallbackHandler; // used only if logging.properties was not found
    private BufferedWriter operationLog;

    public ErrMsgLog() throws IOException {
        Files.createDirectories(Paths.get(LOG_DIR));
        InputStream config = ErrMsgLog.class.getResourceAsStream(CONFIG_RESOURCE);
        Logger logger = Logger.getLogger(ErrMsgLog.class.getName());
        if (config != null) {
            try {
                LogManager.getLogManager().readConfiguration(config);
            } finally {
                config.close();
            }
        } else {
            // the resource was not copied next to the classes: configure the logger in code
            fallbackHandler = new FileHandler(LOG_DIR + "/err_log.txt", true);
            fallbackHandler.setFormatter(new SimpleFormatter());
            logger.setUseParentHandlers(false);
            logger.addHandler(fallbackHandler);
        }
        this.log = logger;
    }

    // ------------------------------------------------------------------ errors

    /** Adds the error to the list of errors. */
    public int addErr(Exception e) {
        errList.add(e);
        return errList.size();
    }

    /** Adds the error to the list of errors and writes it to the log. */
    public int addErrWithLog(Exception e) {
        errList.add(e);
        log.log(Level.WARNING, e.getClass().getSimpleName() + ": " + e.getMessage());
        return errList.size();
    }

    /** Number of registered errors. */
    public int getErrCount() {
        return errList.size();
    }

    /** Shows the text of the error. */
    public void showErrText(Exception e) {
        System.err.println(e.getMessage());
    }

    /**
     * Registers the error (list and log) and returns a new exception that can be thrown
     * to the upper level.
     */
    public Exception makeErr(Exception e) {
        addErrWithLog(e);
        return new Exception(e.getMessage(), e);
    }

    // ------------------------------------------------------------------ operation logs

    /** Starts a new log file with the given name in the log folder. */
    public void startOperationLog(String fileName) throws IOException {
        finishOperationLog();
        operationLog = new BufferedWriter(new OutputStreamWriter(
                Files.newOutputStream(Paths.get(LOG_DIR, fileName)), StandardCharsets.UTF_8));
    }

    /** Writes one line to the current log file. */
    public void writeOperationLine(String line) throws IOException {
        if (operationLog == null) {
            throw new IOException("The log file is not opened");
        }
        operationLog.write(line);
        operationLog.newLine();
    }

    /** Closes the current log file (does nothing if there is none). */
    public void finishOperationLog() throws IOException {
        if (operationLog != null) {
            BufferedWriter writer = operationLog;
            operationLog = null;
            writer.close();
        }
    }

    /** Releases the log files. Call it at the end of the program. */
    public void close() {
        try {
            finishOperationLog();
        } catch (IOException e) {
            System.err.println("Cannot close the log file: " + e.getMessage());
        }
        if (fallbackHandler != null) {
            fallbackHandler.close();
        }
        LogManager.getLogManager().reset(); // closes the handlers of the error log
    }
}

package lab3.controller;

import java.util.Arrays;

/** Execution times (in nanoseconds) of the operations of one type: count, total, average, median. */
public class OperationStats {

    private final String operation;
    private long[] times = new long[16];
    private int count;
    private long total;

    public OperationStats(String operation) {
        this.operation = operation;
    }

    public String getOperation() {
        return operation;
    }

    public void record(long nanos) {
        if (count == times.length) {
            times = Arrays.copyOf(times, count * 2);
        }
        times[count++] = nanos;
        total += nanos;
    }

    public int getCount() {
        return count;
    }

    public long getTotal() {
        return total;
    }

    /** Arithmetic mean of the times (0 if there are no operations). */
    public long getAverage() {
        return count == 0 ? 0L : total / count;
    }

    /** Median of the times (0 if there are no operations). */
    public long getMedian() {
        return median(getTimes());
    }

    /** Times of the operations in the order of their execution. */
    public long[] getTimes() {
        return Arrays.copyOf(times, count);
    }

    /** Median of an array (the array is sorted in place). */
    public static long median(long[] values) {
        if (values.length == 0) {
            return 0L;
        }
        Arrays.sort(values);
        int mid = values.length / 2;
        return values.length % 2 == 1 ? values[mid] : (values[mid - 1] + values[mid]) / 2;
    }
}

package lab3.controller;

import java.util.ArrayList;
import java.util.List;
import java.util.Locale;

/**
 * Checks whether the time of add() in an {@link ArrayList} depends on the moments when the
 * internal array grows. The growth moments are calculated for two rules:
 * <ul>
 * <li>the rule from the lab text: newCapacity = (oldCapacity * 3) / 2 + 1 (Java 6);</li>
 * <li>the rule of modern JDK (7 and later): newCapacity = oldCapacity + (oldCapacity / 2).</li>
 * </ul>
 * In both cases the first add() allocates an array of 10 elements. The very first add() is
 * excluded from the comparison because its time is mostly the cost of the first call.
 */
public class ArrayListGrowthAnalyzer {

    private static final int DEFAULT_CAPACITY = 10;
    private static final int SHOWN_POINTS = 8;

    /** @return the numbers (starting from 1) of the add() calls that make the array grow */
    public static List<Integer> growthPoints(int size, boolean labFormula) {
        List<Integer> points = new ArrayList<Integer>();
        int capacity = 0;
        for (int element = 1; element <= size; element++) {
            if (element > capacity) {
                if (capacity == 0) {
                    capacity = DEFAULT_CAPACITY;
                } else if (labFormula) {
                    capacity = (capacity * 3) / 2 + 1;
                } else {
                    capacity = capacity + (capacity >> 1);
                }
                points.add(element);
            }
        }
        return points;
    }

    /**
     * Compares the times of the add() calls that cause growth with the times of all other calls.
     *
     * @param addTimes times of all add() calls in the order of execution (nanoseconds)
     * @return lines of the report
     */
    public static List<String> analyze(long[] addTimes) {
        List<String> lines = new ArrayList<String>();
        lines.add("ArrayList, " + addTimes.length + " elements: add() time at the moments of array growth");
        if (addTimes.length < 2) {
            lines.add("  not enough data");
            return lines;
        }
        lines.add(describe(addTimes, true));
        lines.add(describe(addTimes, false));
        return lines;
    }

    private static String describe(long[] times, boolean labFormula) {
        List<Integer> points = growthPoints(times.length, labFormula);
        boolean[] growth = new boolean[times.length];
        int growthCount = 0;
        for (int point : points) {
            if (point > 1) { // skip the first add(): it only creates the array
                growth[point - 1] = true;
                growthCount++;
            }
        }
        long[] growthTimes = new long[growthCount];
        long[] otherTimes = new long[times.length - 1 - growthCount];
        int g = 0;
        int o = 0;
        for (int i = 1; i < times.length; i++) {
            if (growth[i]) {
                growthTimes[g++] = times[i];
            } else {
                otherTimes[o++] = times[i];
            }
        }
        StringBuilder sb = new StringBuilder();
        sb.append(labFormula ? "  Rule (old*3)/2+1" : "  Rule old+old/2  ");
        sb.append(": growth at add #");
        for (int i = 1; i < points.size() && i <= SHOWN_POINTS; i++) {
            sb.append(i > 1 ? ", " : "").append(points.get(i));
        }
        sb.append(points.size() - 1 > SHOWN_POINTS ? ", ..." : "");
        sb.append(" (").append(growthCount).append(" times)");
        if (growthCount == 0 || otherTimes.length == 0) {
            sb.append(": not enough data");
            return sb.toString();
        }
        long growthMedian = OperationStats.median(growthTimes);
        long otherMedian = OperationStats.median(otherTimes);
        sb.append(String.format(Locale.US,
                "%n      growth adds: average %d ns, median %d ns; other adds: average %d ns, median %d ns;"
                        + " median ratio = %.1f",
                average(growthTimes), growthMedian, average(otherTimes), otherMedian,
                otherMedian == 0 ? 0.0 : (double) growthMedian / otherMedian));
        return sb.toString();
    }

    private static long average(long[] values) {
        long sum = 0;
        for (long v : values) {
            sum += v;
        }
        return values.length == 0 ? 0L : sum / values.length;
    }
}
