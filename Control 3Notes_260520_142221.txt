#include <iostream>
#include <cstring>

using namespace std;

// Forward declarations
class Product;
class Category;
class Customer;

// ----------------------------- Product -----------------------------
class Product {
private:
int code;
char* name;
float baseCost;
public:
Product();
Product(int c, const char* n, float cost);
~Product();
Product(const Product& other);
Product& operator=(const Product& other);

int getCode() const;
const char* getName() const;
float getBaseCost() const;

void setCode(int c);
void setName(const char* n);
void setBaseCost(float cost);
};

// ----------------------------- Category -----------------------------
class Category {
private:
int code;
char* name;
public:
Category();
Category(int c, const char* n);
~Category();
Category(const Category& other);
Category& operator=(const Category& other);

int getCode() const;
const char* getName() const;

void setCode(int c);
void setName(const char* n);
};

// ----------------------------- Customer (base class) -----------------------------
class Customer {
protected:
char* firstName;
char* lastName;
char* phone;
float baseDiscount; // in percent
public:
Customer();
Customer(const char* fname, const char* lname, const char* ph, float discount);
virtual ~Customer();
Customer(const Customer& other);
Customer& operator=(const Customer& other);

// Getters
const char* getFirstName() const;
const char* getLastName() const;
const char* getPhone() const;
float getBaseDiscount() const;

// Setters
void setFirstName(const char* fname);
void setLastName(const char* lname);
void setPhone(const char* ph);
void setBaseDiscount(float discount);

// Pure virtual methods for polymorphism
virtual float calculateDiscount(const Product& product, const Category& category) = 0;
virtual float calculateFinalCost(const Product& product, const Category& category) = 0;
virtual const char* getCustomerType() const = 0;
};

// ----------------------------- RegularCustomer -----------------------------
class RegularCustomer : public Customer {
public:
RegularCustomer();
RegularCustomer(const char* fname, const char* lname, const char* ph, float discount);
~RegularCustomer();

float calculateDiscount(const Product& product, const Category& category) override;
float calculateFinalCost(const Product& product, const Category& category) override;
const char* getCustomerType() const override;
};

// ----------------------------- PensionerCustomer -----------------------------
class PensionerCustomer : public Customer {
private:
int discountStartHour;
int discountEndHour;
float extraDiscount; // percent
public:
PensionerCustomer();
PensionerCustomer(const char* fname, const char* lname, const char* ph, float discount,
int startHour, int endHour, float extraDisc);
~PensionerCustomer();

float calculateDiscount(const Product& product, const Category& category) override;
float calculateFinalCost(const Product& product, const Category& category) override;
const char* getCustomerType() const override;

// Getters for extra data
int getStartHour() const;
int getEndHour() const;
float getExtraDiscount() const;
};

// ----------------------------- VIPCustomer -----------------------------
class VIPCustomer : public Customer {
private:
int* favoriteProducts; // array of product codes
int favoriteCount;
float extraDiscount; // percent on favorite products
public:
VIPCustomer();
VIPCustomer(const char* fname, const char* lname, const char* ph, float discount,
int* favProds, int count, float extraDisc);
~VIPCustomer();
VIPCustomer(const VIPCustomer& other);
VIPCustomer& operator=(const VIPCustomer& other);

float calculateDiscount(const Product& product, const Category& category) override;
float calculateFinalCost(const Product& product, const Category& category) override;
const char* getCustomerType() const override;

bool isFavoriteProduct(int productCode) const;
int getFavoriteCount() const;
float getExtraDiscount() const;
};

// ----------------------------- Implementation of Product -----------------------------
Product::Product() : code(0), name(nullptr), baseCost(0.0f) {}

Product::Product(int c, const char* n, float cost) : code(c), baseCost(cost) {
name = new char[strlen(n) + 1];
strcpy(name, n);
}

Product::~Product() {
delete[] name;
}

Product::Product(const Product& other) : code(other.code), baseCost(other.baseCost) {
name = new char[strlen(other.name) + 1];
strcpy(name, other.name);
}

Product& Product::operator=(const Product& other) {
if (this != &other) {
code = other.code;
baseCost = other.baseCost;
delete[] name;
name = new char[strlen(other.name) + 1];
strcpy(name, other.name);
}
return *this;
}

int Product::getCode() const { return code; }
const char* Product::getName() const { return name; }
float Product::getBaseCost() const { return baseCost; }

void Product::setCode(int c) { code = c; }
void Product::setName(const char* n) {
delete[] name;
name = new char[strlen(n) + 1];
strcpy(name, n);
}
void Product::setBaseCost(float cost) { baseCost = cost; }

// ----------------------------- Implementation of Customer -----------------------------
Customer::Customer() : firstName(nullptr), lastName(nullptr), phone(nullptr), baseDiscount(0.0f) {}

Customer::Customer(const char* fname, const char* lname, const char* ph, float discount) : baseDiscount(discount) {
firstName = new char[strlen(fname) + 1];
lastName = new char[strlen(lname) + 1];

firstName = new char[strlen(other.firstName) + 1];
lastName = new char[strlen(other.lastName) + 1];
phone = new char[strlen(other.phone) + 1];
strcpy(firstName, other.firstName);
strcpy(lastName, other.lastName);
strcpy(phone, other.phone);
}

Customer& Customer::operator=(const Customer& other) {
if (this != &other) {
baseDiscount = other.baseDiscount;
delete[] firstName; delete[] lastName; delete[] phone;
firstName = new char[strlen(other.firstName) + 1];
lastName = new char[strlen(other.lastName) + 1];
phone = new char[strlen(other.phone) + 1];
strcpy(firstName, other.firstName);
strcpy(lastName, other.lastName);
strcpy(phone, other.phone);
}
return *this;
}

// ----------------------------- RegularCustomer -----------------------------
RegularCustomer::RegularCustomer() : Customer() {}
RegularCustomer::RegularCustomer(const char* fname, const char* lname, const char* ph, float discount)
: Customer(fname, lname, ph, discount) {}
RegularCustomer::~RegularCustomer() {}

float RegularCustomer::calculateDiscount(const Product& product, const Category& category) {
// No extra discount, just base
return getBaseDiscount();
}

float RegularCustomer::calculateFinalCost(const Product& product, const Category& category) {
float discountPercent = calculateDiscount(product, category);
float finalCost = product.getBaseCost() * (1.0f - discountPercent / 100.0f);
return finalCost;
}

const char* RegularCustomer::getCustomerType() const {
return "Regular";
}

// ----------------------------- PensionerCustomer -----------------------------
PensionerCustomer::PensionerCustomer() : Customer(), discountStartHour(0), discountEndHour(0), extraDiscount(0) {}
PensionerCustomer::PensionerCustomer(const char* fname, const char* lname, const char* ph, float discount,
int startHour, int endHour, float extraDisc)
: Customer(fname, lname, ph, discount), discountStartHour(startHour), discountEndHour(endHour), extraDiscount(extraDisc) {}
PensionerCustomer::~PensionerCustomer() {}

float PensionerCustomer::calculateDiscount(const Product& product, const Category& category) {
time_t now = time(0);
tm* ltm = localtime(&now);
int currentHour = ltm->tm_hour;

float totalDiscount = getBaseDiscount();
if (currentHour >= discountStartHour && currentHour < discountEndHour) {
totalDiscount += extraDiscount;
}
return totalDiscount;
}

float PensionerCustomer::calculateFinalCost(const Product& product, const Category& category) {
float discountPercent = calculateDiscount(product, category);
return product.getBaseCost() * (1.0f - discountPercent / 100.0f);
}

const char* PensionerCustomer::getCustomerType() const {
return "Pensioner";
}

int PensionerCustomer::getStartHour() const { return discountStartHour; }
int PensionerCustomer::getEndHour() const { return discountEndHour; }
float PensionerCustomer::getExtraDiscount() const { return extraDiscount; }

// ----------------------------- VIPCustomer -----------------------------
VIPCustomer::VIPCustomer() : Customer(), favoriteProducts(nullptr), favoriteCount(0), extraDiscount(0) {}
VIPCustomer::VIPCustomer(const char* fname, const char* lname, const char* ph, float discount,
int* favProds, int count, float extraDisc)
: Customer(fname, lname, ph, discount), favoriteCount(count), extraDiscount(extraDisc) {
favoriteProducts = new int[count];
for (int i = 0; i < count; i++) {
favoriteProducts[i] = favProds[i];
}
}
VIPCustomer::~VIPCustomer() {
delete[] favoriteProducts;
}
VIPCustomer::VIPCustomer(const VIPCustomer& other) : Customer(other), favoriteCount(other.favoriteCount), extraDiscount(other.extraDiscount) {
favoriteProducts = new int[favoriteCount];
for (int i = 0; i < favoriteCount; i++) {
favoriteProducts[i] = other.favoriteProducts[i];
}
}
VIPCustomer& VIPCustomer::operator=(const VIPCustomer& other) {
if (this != &other) {
Customer::operator=(other);
delete[] favoriteProducts;
favoriteCount = other.favoriteCount;
extraDiscount = other.extraDiscount;
favoriteProducts = new int[favoriteCount];
for (int i = 0; i < favoriteCount; i++) {
favoriteProducts[i] = other.favoriteProducts[i];
}
}
return *this;
}

bool VIPCustomer::isFavoriteProduct(int productCode) const {
for (int i = 0; i < favoriteCount; i++) {
if (favoriteProducts[i] == productCode) return true;
}
return false;
}

float VIPCustomer::calculateDiscount(const Product& product, const Category& category) {
float totalDiscount = getBaseDiscount();
if (isFavoriteProduct(product.getCode())) {
totalDiscount += extraDiscount;
}
return totalDiscount;
}

float VIPCustomer::calculateFinalCost(const Product& product, const Category& category) {
float discountPercent = calculateDiscount(product, category);
return product.getBaseCost() * (1.0f - discountPercent / 100.0f);
}

const char* VIPCustomer::getCustomerType() const {
return "VIP";
}

int VIPCustomer::getFavoriteCount() const { return favoriteCount; }
float VIPCustomer::getExtraDiscount() const { return extraDiscount; }

// ----------------------------- Helper function to get current time as string -----------------------------
string getCurrentDateTime() {
time_t now = time(0);
tm* ltm = localtime(&now);
char buffer[80];
strftime(buffer, sizeof(buffer), "%Y-%m-%d %H:%M:%S", ltm);
return string(buffer);
}
